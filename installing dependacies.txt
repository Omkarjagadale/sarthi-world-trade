#installing dependacies
echo"################################"
echo"installing pkg"
echo"################################"
sudo yum install httpd wget unzip -y
echo

#Start and enable HTTPD services
echo"################################"
echo"Start and enable HTTPD services"
echo"################################"
systemctl start httpd
systemctl enable httpd
echo

# Create temp directory
echo"################################"
echo"Starting artifact deployment"
echo"################################"
mkdir -p /tmp/webfile
cd /tmp/webfile
echo

#download template for the website
wget https://www.tooplate.com/zip-templates/2124_vertex.zip
unzip -o 2124_vertex.zip
sudo cp -r 2124_vertex/* /var/www/html/
echo

#restart httpd
echo"################################"
echo"Restart HTTPD services"
echo"################################"
systemctl restart httpd
echo

#clean up
rm -rf /temp/webfile
